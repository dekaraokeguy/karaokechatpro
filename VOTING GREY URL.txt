"use strict";



console.log("✅ voting loaded");

window.renderContestants = function(){



  if(!window.contestantList) return;



  window.contestantList.innerHTML = "";



  if(!window.contestants.length){



    window.contestantList.innerHTML =

      "<div style='padding:20px;text-align:center;'>No contestants yet</div>";



    return;

  }



  window.contestants.forEach(name=>{



    const row =

      document.createElement("div");



    row.className =

      "leaderboardRow";



    row.innerHTML = `



      <div style="

        font-weight:bold;

        color:white;

        flex:1;

      ">

        🎤 ${name}

      </div>



      <select

        id="vote_${name}"

window.submitVote = function(name){



  const select =

    document.getElementById(

      "vote_" + name

    );



  if(!select) return;



  const score =

    Number(select.value);



  socket.emit(

    "submit people vote",

    {

      contestant:name,

      score:score

    }

  );



  alert(

    "✅ Vote submitted for " +

    name

  );



};
